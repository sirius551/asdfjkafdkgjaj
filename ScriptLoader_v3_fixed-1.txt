local Conns, Cleanups, AccentObjects = {}, {}, {}

local function Add(c) table.insert(Conns, c) return c end
local function OnUnload(fn) table.insert(Cleanups, fn) end

local function Round(o, r) return New("UICorner", {CornerRadius = UDim.new(0, r or 6)}, o) end

local function Tween(o, t, props, style, dir)
    local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function BindAccent(inst, prop)
    table.insert(AccentObjects, {inst, prop})
    inst[prop] = Theme.Accent
end

local function GetChar() return LocalPlayer.Character end
local function GetHum()
    local c = GetChar()
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function GetRoot()
    local c = GetChar()
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function nearestEnemy(maxDist, teamCheck)
    local root = GetRoot()
    if not root then return nil end
    local best, bd, bhum = nil, maxDist or math.huge, nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and not (teamCheck and p.Team ~= nil and p.Team == LocalPlayer.Team) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local tr = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and tr then
                local d = (tr.Position - root.Position).Magnitude
                if d < bd then bd, best, bhum = d, p, hum end
            end
        end
    end
    return best, bd, bhum
end


-- Безопасный клик: на мобильных (Delta) mouse1click может крашить, поэтому там используем Tool:Activate / VirtualUser
local function ClickOnce()
    local c = LocalPlayer.Character
    local tool = c and c:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
        return
    end
    if not isMobile and mouse1click then
        pcall(mouse1click)
        return
    end
    pcall(function()
        local vu = game:GetService("VirtualUser")
        vu:CaptureController()
        vu:Button1Down(Vector2.new(0, 0), workspace.CurrentCamera.CFrame)
        task.wait()
        vu:Button1Up(Vector2.new(0, 0), workspace.CurrentCamera.CFrame)
    end)
end

local BgObjects = {}  -- {inst, prop, darkVal, lightVal}
local isLightTheme = false

local THEMES = {
    dark = {
        Background   = Color3.fromRGB(20, 20, 26),
        Top          = Color3.fromRGB(28, 28, 37),
        Sidebar      = Color3.fromRGB(24, 24, 31),
        Element      = Color3.fromRGB(36, 36, 47),
        ElementHover = Color3.fromRGB(48, 48, 63),
        Text         = Color3.fromRGB(235, 235, 240),
        SubText      = Color3.fromRGB(150, 150, 165),
    },
    light = {
        Background   = Color3.fromRGB(240, 241, 248),
        Top          = Color3.fromRGB(255, 255, 255),
        Sidebar      = Color3.fromRGB(228, 230, 240),
        Element      = Color3.fromRGB(210, 213, 230),
        ElementHover = Color3.fromRGB(195, 198, 220),
        Text         = Color3.fromRGB(25, 25, 40),
        SubText      = Color3.fromRGB(90, 90, 115),
    },
}

local function BindBg(inst, prop, darkVal, lightVal)
    local entry = {inst, prop, darkVal, lightVal}
    table.insert(BgObjects, entry)
    inst[prop] = isLightTheme and lightVal or darkVal
end

local function ApplyFullTheme(light)
    isLightTheme = light
    local t = light and THEMES.light or THEMES.dark
    Theme.Background, Theme.Top = t.Background, t.Top
    Theme.Sidebar, Theme.Element = t.Sidebar, t.Element
    Theme.ElementHover, Theme.Text = t.ElementHover, t.Text
    Theme.SubText = t.SubText
    for i = #BgObjects, 1, -1 do
        local e = BgObjects[i]
        if e[1] and e[1].Parent then
            e[1][e[2]] = light and e[4] or e[3]
        else
            table.remove(BgObjects, i)
        end
    end
end

local function MakeDraggable(handle, target)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, i.Position, target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    Add(UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = (i.Position - dragStart) / math.max(MainScale.Scale, 0.1)
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end))
end

local function FetchSource(url)
    local src
    local ok = pcall(function() src = game:HttpGet(url) end)
    if ok and type(src) == "string" and #src > 0 then return src end
    ok = pcall(function() src = game:HttpGet(url, true) end)
    if ok and type(src) == "string" and #src > 0 then return src end
    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if req then
        local ok2, res = pcall(req, {Url = url, Method = "GET"})
        if ok2 and res and type(res.Body) == "string" and #res.Body > 0 then return res.Body end
    end
    return nil
end

-- Подменяет loadstring внутри запускаемого кода: если скачанный текст не Lua (404, HTML и т.д.),
-- вместо «attempt to call a nil value» будет настоящая причина и начало ответа сервера.
local function wrapEnv(fn)
    pcall(function()
        local base = (getgenv and getgenv()) or getfenv(0)
        local env = setmetatable({
            loadstring = function(src, ...)
                local f, e = loadstring(src, ...)
                if not f then
                    local head = string.gsub(string.sub(tostring(src), 1, 80), "%s+", " ")
                    error("loadstring не смог скомпилировать код: " .. tostring(e) .. " | начало ответа: " .. head, 0)
                end
                return f
            end,
        }, {__index = base, __newindex = base})
        setfenv(fn, env)
    end)
    return fn
end

local function RunUrl(name, url)
    Notify("Загрузка", name .. "...", 1.5)
    local src = FetchSource(url)
    if not src then error("не удалось скачать скрипт (ссылка недоступна?)", 0) end
    src = src:gsub("^\239\187\191", "")
    if string.find(string.sub(src, 1, 300):lower(), "^%s*<") then
        error("ссылка вернула HTML-страницу, а не Lua (скрипт удалён или заблокирован)", 0)
    end
    local fn, cerr = loadstring(src)
    if not fn then error("ошибка компиляции: " .. tostring(cerr), 0) end
    local ok, err = pcall(wrapEnv(fn))
    if not ok then error(tostring(err), 0) end
    Notify("Готово", name .. " запущен", 2)
end

-- Чип с биндом клавиши рядом с функцией
local function AttachBind(parent, id, action, pos)
    Actions[id] = action
    local saved = Config.Binds[id]
    local chip = New("TextButton", {
        BackgroundColor3 = Color3.fromRGB(28, 28, 37), BorderSizePixel = 0, AutoButtonColor = false,
        AnchorPoint = Vector2.new(1, 0.5), Position = pos, Size = UDim2.new(0, 46, 0, 20),
        Font = Enum.Font.GothamBold, TextSize = 9, TextColor3 = Theme.SubText,
        Text = saved and string.sub(saved, 1, 6) or "—", ZIndex = 5,
    }, parent)
    Round(chip, 5)
    chip.MouseButton1Click:Connect(function()
        if Listening then return end
        Listening = true
        chip.Text = "..."
        local c
        c = UIS.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.Keyboard then
                local n = i.KeyCode.Name
                if n == "Escape" or n == "Backspace" then
                    Config.Binds[id] = nil
                    chip.Text = "—"
                else
                    Config.Binds[id] = n
                    chip.Text = string.sub(n, 1, 6)
                end
                SaveConfig()
                c:Disconnect()
                task.delay(0.2, function() Listening = false end)
            end
        end)
    end)
    return chip
end

local function ApplySearch()
    local q = string.lower(SearchBox.Text)
    local tab = Tabs[ActiveTab]
    if not tab then return end
    for _, item in ipairs(tab.Items) do
        if item.IsSection then
            item.Frame.Visible = (q == "")
        else
            item.Frame.Visible = (q == "") or (string.find(string.lower(item.Name), q, 1, true) ~= nil)
        end
    end
end
SearchBox:GetPropertyChangedSignal("Text"):Connect(ApplySearch)

    ActiveTab = name
    for n, t in pairs(Tabs) do t.Page.Visible = (n == name) end

    RefreshTabs()
end

    function tab:Slider(text, min, max, default, callback, decimals)
        decimals = decimals or 0
        local value = math.clamp(default, min, max)
        local f = New("Frame", {BackgroundColor3 = Theme.Element, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 46), LayoutOrder = nextOrder()}, page)
        Round(f, 6)
        New("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(0, 12, 0, 4), Size = UDim2.new(1, -90, 0, 18),
            Font = Enum.Font.GothamSemibold, Text = text, TextColor3 = Theme.Text, TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Left,
        }, f)
        local valLabel = New("TextLabel", {
            BackgroundTransparency = 1, Position = UDim2.new(1, -72, 0, 4), Size = UDim2.new(0, 60, 0, 18),
            Font = Enum.Font.GothamBold, TextColor3 = Theme.SubText, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Right,

            BackgroundColor3 = Color3.fromRGB(55, 55, 70), BorderSizePixel = 0,
            Position = UDim2.new(0, 12, 0, 31), Size = UDim2.new(1, -24, 0, 6),
        }, f)
        Round(track, 3)
        local fill = New("Frame", {BorderSizePixel = 0, Size = UDim2.new(0, 0, 1, 0)}, track)
        BindAccent(fill, "BackgroundColor3")
        Round(fill, 3)

        local function fmt(v)
            local m = 10 ^ decimals
            return math.floor(v * m + 0.5) / m
        end
        local obj = {}
        local function setValue(v, silent)
            value = fmt(math.clamp(v, min, max))
            local a = (value - min) / (max - min)
            fill.Size = UDim2.new(a, 0, 1, 0)
            valLabel.Text = tostring(value)
            if not silent then
                local ok, err = pcall(callback, value)
                if not ok then warn("[ScriptHub Slider Error]: " .. text .. " -> " .. tostring(err)) end
            end
        end
        function obj:Set(v) setValue(v, true) end
        function obj:Get() return value end

        local dragging = false
        local function fromInput(x)
            local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
            setValue(min + (max - min) * a)
        end
        track.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                page.ScrollingEnabled = false
                fromInput(i.Position.X)
            end
        end)
        Add(UIS.InputChanged:Connect(function(i)
            if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
                fromInput(i.Position.X)
            end
        end))
        Add(UIS.InputEnded:Connect(function(i)
            if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
                dragging = false
                page.ScrollingEnabled = true
                if obj.OnRelease then obj.OnRelease(value) end
            end
        end))

        setValue(value, true)
        register(f, text)
        return obj
    end

    return tab
end

local function ApplyAccent(color, save)
    Theme.Accent = color
    Config.Accent = {math.floor(color.R * 255 + 0.5), math.floor(color.G * 255 + 0.5), math.floor(color.B * 255 + 0.5)}
    for i = #AccentObjects, 1, -1 do
        local o = AccentObjects[i]
        if o[1] and o[1].Parent then
            o[1][o[2]] = color
        else
            table.remove(AccentObjects, i)
        end
    end
    for _, t in pairs(Tabs) do t.Page.ScrollBarImageColor3 = color end
    for _, tg in ipairs(Toggles) do tg.Render() end
    RefreshTabs()
    if save then SaveConfig() end
end

local Unloading = false
local function UnloadHub()
    if Unloading then return end
    Unloading = true
    pcall(function()
        Tween(MainFrame, 0.25, {BackgroundTransparency = 1, Size = UDim2.new(0, 0, 0, 0)})
    end)
    task.wait(0.27)
    for _, tg in ipairs(Toggles) do pcall(function() if tg:Get() then tg:Set(false) end end) end
    for _, fn in ipairs(Cleanups) do pcall(fn) end

FlingTab:Button("Fling All", function()
    local function SkidFling(TargetPlayer)
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = TargetPlayer.Character
        if not (TCharacter and Character and Humanoid and RootPart) then return end
        local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter:FindFirstChild("Head")
        if not (TRootPart or THead) then return end

        if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = RootPart.CFrame end

        local function FPos(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end
        local function SFBasePart(BasePart)
            local Time, Angle = tick(), 0
            repeat
                if RootPart and THumanoid then
                    Angle += 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else break end
            until BasePart.Velocity.Magnitude > 500 or BasePart.Parent ~= TargetPlayer.Character or Humanoid.Health <= 0 or tick() > Time + 2
        end

        local oldDestroyHeight = workspace.FallenPartsDestroyHeight
        workspace.FallenPartsDestroyHeight = 0 / 0
        local BV = Instance.new("BodyVelocity")
        BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
        BV.MaxForce = Vector3.new(1 / 0, 1 / 0, 1 / 0)
        BV.Parent = RootPart
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
        if TRootPart then SFBasePart(TRootPart) elseif THead then SFBasePart(THead) end
        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        Camera.CameraSubject = Humanoid
        -- Возврат на место (раньше персонаж оставался в точке финиша)
        pcall(function()
            RootPart.Velocity, RootPart.RotVelocity = Vector3.zero, Vector3.zero
            if getgenv().OldPos then RootPart.CFrame = getgenv().OldPos end
        end)
        workspace.FallenPartsDestroyHeight = oldDestroyHeight
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then pcall(SkidFling, p) end
    end
end)

    local function getClosest()
        local cam = workspace.CurrentCamera
        local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
        local closest, shortest = nil, math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
                if not (aimbot.TeamCheck and p.Team ~= nil and p.Team == LocalPlayer.Team) then
                    local hum = p.Character:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        local head = p.Character.Head
                        local sp, onScreen = cam:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if d <= aimbot.FOV and d < shortest then shortest, closest = d, head end
                        end
                    end
                end
            end
        end
        return closest
    end
    table.insert(aimbot.Conns, UIS.InputBegan:Connect(function(i, gpe)
        if gpe then return end
        if i.KeyCode == Enum.KeyCode.Q or i.UserInputType == Enum.UserInputType.MouseButton2 then aiming = true end
    end))
    table.insert(aimbot.Conns, UIS.InputEnded:Connect(function(i)
        if i.KeyCode == Enum.KeyCode.Q or i.UserInputType == Enum.UserInputType.MouseButton2 then aiming = false end
    end))

        local cam = workspace.CurrentCamera

            aimbot.Circle.Position = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

        end
    end))

RageTab:Section("Bunnyhop")
-- В Roblox зажатый Space и так прыгает без остановки, поэтому главное здесь — разгон скорости.
local bh = {progressive = true, step = 4, limit = 70, fixed = 32, base = 16, grace = 0.15,
    strafe = true, forceJump = false, cur = 16, conn = nil}
local bhLabel = RageTab:Label("Скорость: 16 / 70")
local function bhRefreshLabel()
    bhLabel.Text = ("Скорость: %d / %d"):format(math.floor(bh.cur + 0.5), bh.progressive and bh.limit or bh.fixed)
end
RageTab:Toggle("Bunnyhop (держи Space)", false, function(on)
    if bh.conn then bh.conn:Disconnect() bh.conn = nil end
    bh.cur = bh.base
    bhRefreshLabel()
    if not on then return end
    local wasGrounded, groundTimer, uiTimer = true, 0, 0
    bh.conn = RunService.Heartbeat:Connect(function(dt)
        local h, r = GetHum(), GetRoot()
        if not (h and r) or h.Health <= 0 then return end
        local holding = (UIS:IsKeyDown(Enum.KeyCode.Space) and not UIS:GetFocusedTextBox()) or h.Jump
        local grounded = h.FloorMaterial ~= Enum.Material.Air
        local md = h.MoveDirection

        if grounded then
            groundTimer += dt
            if bh.forceJump and holding then h:ChangeState(Enum.HumanoidStateType.Jumping) end
            -- не успел прыгнуть вовремя или отпустил Space — скорость сбрасывается
            if groundTimer > bh.grace or not holding then bh.cur = bh.base end
        elseif wasGrounded then
            -- момент отрыва от земли = новый прыжок
            groundTimer = 0
            if holding then
                if bh.progressive then
                    bh.cur = math.min(bh.limit, math.max(bh.cur, bh.base) + bh.step)
                else
                    bh.cur = bh.fixed
                end
            else
                bh.cur = bh.base -- просто сошёл с уступа
            end
        end
        wasGrounded = grounded

        if md.Magnitude > 0 and bh.cur > bh.base and (not grounded or groundTimer <= bh.grace) then
            local v = r.AssemblyLinearVelocity
            local horiz = Vector3.new(v.X, 0, v.Z)
            local target = md.Unit * bh.cur
            local nv = grounded and target or (bh.strafe and horiz:Lerp(target, 0.2) or horiz)
            r.AssemblyLinearVelocity = Vector3.new(nv.X, v.Y, nv.Z)
        end

        uiTimer += dt
        if uiTimer > 0.1 then uiTimer = 0 bhRefreshLabel() end
    end)
end)
OnUnload(function() if bh.conn then bh.conn:Disconnect() end end)

    local cam = workspace.CurrentCamera

    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {GetChar(), cam}
    local origin = cam.CFrame.Position
    local res = workspace:Raycast(origin, part.Position - origin, params)
    return res == nil or res.Instance:IsDescendantOf(part.Parent)

                local d = (part.Position - root.Position).Magnitude
                if d < bd and (not rage.wall or visibleCheck(part)) then bd, best = d, part end
            end
        end
    end
    return best
end

                        e.head.Position = Vector2.new(headPos.X, headPos.Y)
                        e.head.Radius, e.head.Color = 4, col
                        e.head.Filled, e.head.Visible = true, VFX.HeadDots
                    end
                    if e.name then
                        e.name.Position = Vector2.new(top.X, y - 15)
                        e.name.Text, e.name.Center = player.DisplayName, true
                        e.name.Size, e.name.Color = 13, col
                        e.name.Outline, e.name.Visible = true, VFX.Names
                    end
                    local ratio = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                    if e.healthBack then
                        e.healthBack.From, e.healthBack.To = Vector2.new(x - 5, y), Vector2.new(x - 5, y + h)
                        e.healthBack.Color, e.healthBack.Thickness = Color3.fromRGB(35, 35, 35), 3
                        e.healthBack.Visible = VFX.HealthBars
                    end
                    if e.health then
                        e.health.From, e.health.To = Vector2.new(x - 5, y + h), Vector2.new(x - 5, y + h * (1 - ratio))
                        e.health.Color, e.health.Thickness = Color3.fromRGB(80, 230, 120), 2
                        e.health.Visible = VFX.HealthBars
                    end
                    local myRoot = GetRoot()
                    local dist = myRoot and math.floor((myRoot.Position - root.Position).Magnitude) or 0
                    if e.distance then
                        e.distance.Position = Vector2.new(top.X, bottom.Y + 2)
                        e.distance.Text, e.distance.Center = tostring(dist) .. "m", true
                        e.distance.Size, e.distance.Color = 12, Color3.new(1, 1, 1)
                        e.distance.Outline, e.distance.Visible = true, VFX.Distances
                    end
                else
                    VFXHideEntry(e)
                end
            end
        end
    end
    for player, e in pairs(VFXHighlights) do
        if typeof(player) == "Instance" and player:IsA("Player") and not active[player] then
            VFXHideEntry(e)
            for _, obj in pairs(e) do pcall(function() obj:Remove() end) end
            VFXHighlights[player] = nil
        end
    end
end

local function VFXBindCharacter(char)
    if VFXCharacterConn then VFXCharacterConn:Disconnect() VFXCharacterConn = nil end
    local hum = char and char:WaitForChild("Humanoid", 5)
    local root = char and char:WaitForChild("HumanoidRootPart", 5)
    if not hum or not root then return end
    local lastState = hum:GetState()
    VFXCharacterConn = hum.StateChanged:Connect(function(_, state)
        if state == Enum.HumanoidStateType.Jumping then
            if VFX.JumpCircle then VFXMakeRing(root.Position - Vector3.new(0, 2.7, 0), 2.4, Theme.Accent, 0.65, 0.14) end
            if VFX.JumpTrail then
                local a = New("Attachment", {}, root)
                local b = New("Attachment", {Position = Vector3.new(0, -1.5, 0)}, root)
                local trail = New("Trail", {
                    Attachment0 = a, Attachment1 = b, Lifetime = 0.35,
                    Color = ColorSequence.new(Theme.Accent), LightEmission = 0.8,
                    Transparency = NumberSequence.new(0.15, 1), WidthScale = NumberSequence.new(0.7, 0),
                }, root)
                task.delay(0.5, function() trail:Destroy() a:Destroy() b:Destroy() end)
            end
        elseif state == Enum.HumanoidStateType.Landed then
            if VFX.LandingRipple then VFXMakeRing(root.Position - Vector3.new(0, 2.7, 0), 3, Color3.fromRGB(130, 200, 255), 0.8, 0.16) end
        end
        lastState = state
    end)
end

FunTab:Section("Локальные эффекты")
local FunLighting = game:GetService("Lighting")
local FunRunService = game:GetService("RunService")
local FunActive = {}
local FunOriginalFov = nil

local function FunCleanup(key)
    local item = FunActive[key]
    if not item then return end
    if key == "FovBreathing" or key == "ZoomWave" then
        local cam = workspace.CurrentCamera
        if cam and FunOriginalFov then cam.FieldOfView = FunOriginalFov end
        FunOriginalFov = nil
    end
    if item.conn then pcall(function() item.conn:Disconnect() end) end
    if item.object then pcall(function() item.object:Destroy() end) end
    if item.objects then
        for _, obj in ipairs(item.objects) do pcall(function() obj:Destroy() end) end
    end
    FunActive[key] = nil
end

local function FunEffectToggle(label, key, create)
    FunTab:Toggle(label, false, function(on)
        FunCleanup(key)
        if on then
            local ok, result = pcall(create)
            if ok and result then
                FunActive[key] = result
            else
                Notify(label, "Не удалось включить эффект: " .. tostring(result), 4, true)
            end
        end
    end)
end

local function FunLightingToggle(label, key, className, properties)
    FunEffectToggle(label, key, function()
        local effect = Instance.new(className)
        effect.Name = "ScriptLoader_Fun_" .. key
        for property, value in pairs(properties) do
            effect[property] = value
        end
        effect.Parent = FunLighting
        return {object = effect}
    end)
end

local playerAccessorySaved = {}
local function SetOwnAccessoriesHidden(hidden)
    local char = GetChar()
    if not char then return end
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Parent and obj.Parent:IsA("Accessory") then
            if hidden then
                if playerAccessorySaved[obj] == nil then playerAccessorySaved[obj] = obj.LocalTransparencyModifier end
                obj.LocalTransparencyModifier = 1
            elseif playerAccessorySaved[obj] ~= nil then
                obj.LocalTransparencyModifier = playerAccessorySaved[obj]
                playerAccessorySaved[obj] = nil
            end
        end
    end
end

local killCount = 0
local killCountConns = {}
local function hookKillCount(p)
    if p == LocalPlayer then return end
    local function hookChar(char)
        local h = char and char:FindFirstChildOfClass("Humanoid")
        if not h then return end
        local c = h.Died:Connect(function() killCount += 1 end)
        table.insert(killCountConns, c)
    end
    if p.Character then hookChar(p.Character) end
    local c = p.CharacterAdded:Connect(hookChar)
    table.insert(killCountConns, c)
end
for _, p in ipairs(Players:GetPlayers()) do hookKillCount(p) end
local killPlayerConn = Players.PlayerAdded:Connect(hookKillCount)

    killPlayerConn:Disconnect()

local emotePlaying = nil
local emoteLoop = false
local emoteLoopConn = nil

local function stopCurrentEmote()
    if emoteLoopConn then emoteLoopConn:Disconnect() emoteLoopConn = nil end
    if emotePlaying then
        pcall(function() emotePlaying:Stop(0.2) end)
        emotePlaying = nil
    end
end
OnUnload(stopCurrentEmote)

local function playEmoteById(assetId)
    stopCurrentEmote()
    local c = GetChar()
    local h = c and c:FindFirstChildOfClass("Humanoid")
    if not h then error("персонаж не найден", 0) end
    local AnimCtrl = h.Animator or h
    local anim = Instance.new("Animation")
    anim.AnimationId = assetId
    local track = AnimCtrl:LoadAnimation(anim)
    track:Play()
    emotePlaying = track
    if emoteLoop then
        emoteLoopConn = track.Stopped:Connect(function()
            if emoteLoop then
                track:Play()
            end
        end)
    end
    return track
end

local function playEmoteByName(name)
    -- Пробуем через LocalScript Animate (FE-безопасный способ)
    local c = GetChar()
    if not c then error("нет персонажа", 0) end
    local animate = c:FindFirstChild("Animate")
    if animate then
        -- Метод 1: через скрипт Animate (работает в большинстве игр)
        local emoteEvent = animate:FindFirstChild("PlayEmote")
        if emoteEvent and emoteEvent:IsA("RemoteEvent") then
            pcall(function() emoteEvent:FireServer(name) end)
            return
        end
        -- Метод 2: имитация /e команды через чат
        pcall(function()
            game:GetService("ReplicatedStorage")
                :WaitForChild("DefaultChatSystemChatEvents", 2)
                :WaitForChild("SayMessageRequest", 2)
                :FireServer("/e " .. name, "All")
        end)
        return
    end
    error("Animate не найден, попробуй эмоцию по ID", 0)
end

local fc = {conn = nil, touchConn = nil, speed = 60, gui = nil, up = false, down = false}
local function StopFreecam()
    if fc.conn then fc.conn:Disconnect() fc.conn = nil end
    if fc.touchConn then fc.touchConn:Disconnect() fc.touchConn = nil end
    if fc.gui then fc.gui:Destroy() fc.gui = nil end
    fc.up, fc.down = false, false
    UIS.MouseBehavior = Enum.MouseBehavior.Default
    local cam = workspace.CurrentCamera
    cam.CameraType = Enum.CameraType.Custom
    local h, r = GetHum(), GetRoot()
    if h then cam.CameraSubject = h end
    if r then r.Anchored = false end
end
UtilTab:Toggle("Freecam", false, function(on)
    StopFreecam()
    if not on then return end
    local cam = workspace.CurrentCamera
    local pitch, yaw = cam.CFrame:ToEulerAnglesYXZ()
    local pos = cam.CFrame.Position
    cam.CameraType = Enum.CameraType.Scriptable
    local r0 = GetRoot()
    if r0 then r0.Anchored = true end

    if isMobile then
        fc.gui = New("Frame", {
            BackgroundTransparency = 1, AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14, 0.5, 0),
            Size = UDim2.new(0, 56, 0, 120), ZIndex = 45,
        }, ScreenGui)
        local function mk(txt, y, key)
            local b = New("TextButton", {
                BackgroundColor3 = Theme.Top, Text = txt, Font = Enum.Font.GothamBold, TextSize = 20,
                TextColor3 = Theme.Text, Position = UDim2.new(0, 0, 0, y), Size = UDim2.new(1, 0, 0, 56),
                AutoButtonColor = false, ZIndex = 46,
            }, fc.gui)
            Round(b, 10)
            b.InputBegan:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then fc[key] = true end
            end)
            b.InputEnded:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then fc[key] = false end
            end)
        end
        mk("▲", 0, "up")
        mk("▼", 64, "down")
        fc.touchConn = UIS.InputChanged:Connect(function(i, gpe)
            if gpe then return end
            if i.UserInputType == Enum.UserInputType.Touch then
                yaw -= i.Delta.X * 0.006
                pitch = math.clamp(pitch - i.Delta.Y * 0.006, -1.5, 1.5)
            end
        end)
    end

    fc.conn = RunService.RenderStepped:Connect(function(dt)
        local rot = CFrame.fromEulerAnglesYXZ(pitch, yaw, 0)
        if isMobile then
            local h = GetHum()
            local md = h and h.MoveDirection or Vector3.zero
            local vy = (fc.up and 1 or 0) - (fc.down and 1 or 0)
            pos += (Vector3.new(md.X, 0, md.Z) + Vector3.new(0, vy, 0)) * fc.speed * dt
        else
            if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                UIS.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
                local d = UIS:GetMouseDelta()
                yaw -= d.X * 0.004
                pitch = math.clamp(pitch - d.Y * 0.004, -1.5, 1.5)
            else
                UIS.MouseBehavior = Enum.MouseBehavior.Default
            end
            rot = CFrame.fromEulerAnglesYXZ(pitch, yaw, 0)
            local mv = Vector3.zero
            if UIS:IsKeyDown(Enum.KeyCode.W) then mv += Vector3.new(0, 0, -1) end
            if UIS:IsKeyDown(Enum.KeyCode.S) then mv += Vector3.new(0, 0, 1) end
            if UIS:IsKeyDown(Enum.KeyCode.A) then mv += Vector3.new(-1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.D) then mv += Vector3.new(1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.E) then mv += Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.Q) then mv += Vector3.new(0, -1, 0) end
            local spd = fc.speed * (UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 3 or 1)
            if mv.Magnitude > 0 then pos += rot:VectorToWorldSpace(mv.Unit) * spd * dt end
        end
        cam.CFrame = CFrame.new(pos) * rot
    end)
end)
OnUnload(StopFreecam)

local function playSound(assetId)
    local snd = New("Sound", {
        SoundId = "rbxassetid://" .. tostring(assetId),
        Volume = 1, RollOffMaxDistance = 9999,
    }, workspace.CurrentCamera)
    snd:Play()
    snd.Ended:Connect(function() snd:Destroy() end)
    task.delay(10, function() if snd and snd.Parent then snd:Destroy() end end)
end

local function getCurrentKsId()
    if ksCustomId ~= "" then return ksCustomId end
    return KS_LIST[ksIdx] and KS_LIST[ksIdx].id or ""
end
local function getCurrentDsId()
    if dsCustomId ~= "" then return dsCustomId end
    return DS_LIST[dsIdx] and DS_LIST[dsIdx].id or ""
end

-- Kill Sound: срабатывает когда умирает кто-то другой (Humanoid.Died)
local killSoundHooked = {}
local function hookKillSound(char)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or killSoundHooked[hum] then return end
    killSoundHooked[hum] = true
    hum.Died:Connect(function()
        if ksActive then
            local id = getCurrentKsId()
            if id ~= "" then playSound(id) end
        end
    end)
end

local ksSoundConn
local ksBtnLabel, dsBtnLabel
local ksSoundToggle
ksSoundToggle = FunTab:Toggle(Lang.killSound, false, function(on)
    ksActive = on
    if on then
        killSoundHooked = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then hookKillSound(p.Character) end
            p.CharacterAdded:Connect(function(c) hookKillSound(c) end)
        end
        if ksSoundConn then ksSoundConn:Disconnect() end
        ksSoundConn = Players.PlayerAdded:Connect(function(p)
            p.CharacterAdded:Connect(function(c) hookKillSound(c) end)
        end)
        Notify(Lang.killSound, Lang.soundOn, 1.5)
    else
        Notify(Lang.killSound, Lang.soundOff, 1.5)
    end
end)
OnUnload(function() if ksSoundConn then ksSoundConn:Disconnect() end ksActive = false end)

local currentSky = nil
local skyFaces = {"SkyboxBk","SkyboxDn","SkyboxFt","SkyboxLf","SkyboxRt","SkyboxUp"}

local function applySky(assetId)
    -- Удаляем старое небо
    for _, obj in ipairs(Lighting:GetChildren()) do
        if obj:IsA("Sky") then obj:Destroy() end
    end
    if not assetId or tostring(assetId) == "" then return end
    local sky = New("Sky", {}, Lighting)
    local texUrl = "rbxassetid://" .. tostring(assetId)
    for _, face in ipairs(skyFaces) do
        sky[face] = texUrl
    end
    currentSky = sky
    return sky
end

local skyBtnRef

skyBtnRef = FunTab:Button("Небо: Pink Sky (нажми для смены)", function()

    local function getCurrentIdx()
        for i, sk in ipairs(SKY_LIST) do
            if skyBtnRef.Text:find(sk.name, 1, true) then return i end
        end
        return 0
    end
    local idx = getCurrentIdx() % #SKY_LIST + 1
    skyBtnRef.Text = "Небо: " .. SKY_LIST[idx].name .. " (нажми для смены)"
    skyCustomId = ""
    applySky(SKY_LIST[idx].id)
    Notify(Lang.skySec, SKY_LIST[idx].name, 2)
end, true)

    local states = {Enum.HumanoidStateType.FallingDown, Enum.HumanoidStateType.Ragdoll}

        end
    end
    if not on then setStates(true) return end
    arConn = RunService.Heartbeat:Connect(function()

        if not h then return end
        setStates(false)
        local st = h:GetState()

            h:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
    end)

    local h = GetHum()

        pcall(function() h:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true) end)
        pcall(function() h:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true) end)
    end
end)

local cd = {conn = nil, stack = {}}
local function UndoDelete()
    local item = table.remove(cd.stack)
    if item and item[1] then item[1].Parent = item[2] end
    return item ~= nil
end
WorldTab:Toggle("Click Delete (тап удаляет объект, локально)", false, function(on)
    if cd.conn then cd.conn:Disconnect() cd.conn = nil end
    if not on then return end
    local mouse = LocalPlayer:GetMouse()
    cd.conn = UIS.InputBegan:Connect(function(i, gpe)
        if gpe then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            local t, c = mouse.Target, GetChar()
            if t and not t:IsA("Terrain") and not (c and t:IsDescendantOf(c)) then
                table.insert(cd.stack, {t, t.Parent})
                t.Parent = nil
            end
        end
    end)
end)
WorldTab:Button("↩ Вернуть удалённый объект", function()
    if not UndoDelete() then error("нечего возвращать", 0) end
end)
OnUnload(function()
    if cd.conn then cd.conn:Disconnect() end
    while UndoDelete() do end
end)

local function tryServerTime(t)
    local found = false
    for _, v in ipairs(game:GetDescendants()) do
        if v:IsA("RemoteEvent") then
            local n = string.lower(v.Name)
            if n:find("time") or n:find("day") or n:find("night") or n:find("light") or n:find("clock") then
                pcall(function() v:FireServer(t) end)
                pcall(function() v:FireServer("SetTime", t) end)
                pcall(function() v:FireServer({time = t}) end)
                found = true
            end
        end
    end
    return found
end

local auraConn
WorldTab:Toggle("Аура-кольцо вокруг персонажа", false, function(on)
    if auraConn then auraConn:Disconnect() auraConn = nil end
    local old = workspace:FindFirstChild("HubAura")
    if old then old:Destroy() end
    if not on then return end
    local folder = New("Folder", {Name = "HubAura"}, workspace)
    local N, parts = 16, {}
    for i = 1, N do
        local part = New("Part", {
            Size = Vector3.new(0.5, 0.5, 0.5), Anchored = true, CanCollide = false,
            CanTouch = false, CanQuery = false, CastShadow = false, Material = Enum.Material.Neon,
            Color = Color3.fromHSV((i - 1) / N, 1, 1), Transparency = 0.1,
        }, folder)
        table.insert(parts, part)
    end
    local t = 0
    auraConn = RunService.Heartbeat:Connect(function(dt)
        local r = GetRoot()
        if not r then return end
        t += dt * 2
        for i, part in ipairs(parts) do
            local angle = (i / N) * math.pi * 2 + t
            local rad = 3 + math.sin(t * 1.5 + i) * 0.5
            part.CFrame = CFrame.new(
                r.Position.X + math.cos(angle) * rad,
                r.Position.Y + 1 + math.sin(t + i * 0.4) * 0.5,

local codeBox = ExecTab:CodeBox("-- вставь Lua-код сюда", 150)
local execStatus = ExecTab:Label("Статус: ожидание")

local function runCode(code)
    code = code or codeBox.Text
    if string.gsub(code, "%s", "") == "" then error("поле кода пустое", 0) end
    local fn, err = loadstring(code)
    if not fn then
        execStatus.Text = Lang.statusSynErr or "Syntax error"
        error("синтаксис: " .. tostring(err), 0)
    end
    local ok, e = pcall(wrapEnv(fn))
    if not ok then
        execStatus.Text = Lang.statusRunErr or "Runtime error"
        error(tostring(e), 0)
    end
    execStatus.Text = "Статус: выполнено (" .. (function() local t=os.time() return string.format("%02d:%02d:%02d",math.floor(t/3600)%24,math.floor(t/60)%60,t%60) end)() .. ")"
    Notify("Executor", "Код выполнен", 2)
end
